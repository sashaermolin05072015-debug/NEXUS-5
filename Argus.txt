/*

ARGUS — full build

ESP32 + ST7735 (128x160) + microSD + IR + Wi-Fi + Buzzer

TFT: CS=15 RST=4 DC=2 BL=16 MOSI=23 SCK=18

SD : CS=5  MOSI=23 MISO=19 SCK=18

BTN: UP=26 SELECT=25 BACK=14 (INPUT_PULLUP)

BUZZER=32  IR=27

Все обращения к SD/TFT идут через один SPI (SCK=18, MOSI=23, MISO=19).

Перед обращением к TFT -> SD_CS HIGH. Перед обращением к SD -> TFT_CS HIGH.
*/

void toneRaw(uint16_t freq, uint16_t dur) {
if (!buzzerEnabled || freq == 0) { delay(dur); return; }
ledcWriteTone(BUZZER_PIN, freq);
delay(dur);
ledcWriteTone(BUZZER_PIN, 0);
}

void applyBrightness() {
if (brightnessLevel < 0) brightnessLevel = 0;

// ================= BUTTONS =================
bool backPressed() {
if (digitalRead(BTN_BACK) == LOW && millis() - lastButton[0] > DEBOUNCE_MS) {
lastButton[0] = millis(); beepBack(); return true;
}
return false;
}
bool nextPressed() {
if (digitalRead(BTN_NEXT) == LOW && millis() - lastButton[1] > DEBOUNCE_MS) {
lastButton[1] = millis(); beepNav(); return true;
}
return false;
}
bool selectPressed() {
if (digitalRead(BTN_SELECT) == LOW && millis() - lastButton[2] > DEBOUNCE_MS) {
lastButton[2] = millis(); beepSelect(); return true;
}
return false;
}

void truncateTo(const char* src, char* out, int maxChars) {
int len = strlen(src);
if (len <= maxChars) { strcpy(out, src); return; }
if (maxChars <= 2) { strncpy(out, src, maxChars); out[maxChars] = '\0'; return; }
strncpy(out, src, maxChars - 2);
out[maxChars - 2] = '.'; out[maxChars - 1] = '.'; out[maxChars] = '\0';
}

void sendIRCommand(const IRCommand& c) {
uint32_t now = millis();
if ((uint32_t)(now - lastIrSend) < IR_SEND_GUARD_MS) return;
switch (c.protocol) {
case IR_PROTO_NEC:        irsend.sendNEC(c.code,c.bits,c.repeat); break;
case IR_PROTO_SAMSUNG:    irsend.sendSAMSUNG(c.code,c.bits,c.repeat); break;
case IR_PROTO_LG:         irsend.sendLG(c.code,c.bits,c.repeat); break;
case IR_PROTO_SONY:       irsend.sendSony(c.code,c.bits,c.repeat); break;
case IR_PROTO_PANASONIC:  irsend.sendPanasonic64(c.code,c.bits,c.repeat); break;
case IR_PROTO_SHARP:      irsend.sendSharpRaw(c.code,c.bits,c.repeat); break;
case IR_PROTO_RC5:        irsend.sendRC5(c.code,c.bits,c.repeat); break;
default: return;
}
lastIrSend = millis();
playIRCue();
}

void sdEnsureDirs() {
spiToSD();
if (!sd.exists("/ARGUS"))          sd.mkdir("/ARGUS");
if (!sd.exists("/ARGUS/GIF"))      sd.mkdir("/ARGUS/GIF");
if (!sd.exists("/ARGUS/IMAGE"))    sd.mkdir("/ARGUS/IMAGE");
if (!sd.exists("/ARGUS/SOUND"))    sd.mkdir("/ARGUS/SOUND");
if (!sd.exists("/ARGUS/CONFIG"))   sd.mkdir("/ARGUS/CONFIG");
if (!sd.exists("/ARGUS/CONFIG/IR"))sd.mkdir("/ARGUS/CONFIG/IR");
spiIdle();
}
void sdListFiles(const char* path, char list[][32], int& count, int maxItems) {
count = 0;
spiToSD();
File dir = sd.open(path);
if (!dir) { spiIdle(); return; }
while (count < maxItems) {
File e = dir.openNextFile();
if (!e) break;
if (e.name()[0] != '.') {
strncpy(list[count], e.name(), 31); list[count][31] = '\0';
count++;
}
e.close();
}
dir.close();
spiIdle();
}
void sdDelete(const char* fullpath) {
spiToSD();
sd.remove(fullpath);
spiIdle();
}
uint32_t sdFreeMB() {
spiToSD();
uint32_t mb = 0;
if (sdReady) mb = (uint32_t)((uint64_t)(sd.vol()->freeClusterCount()) * sd.vol()->blocksPerCluster() * 512ULL / (1024ULL*1024ULL));
spiIdle();
return mb;
}

// ================= CONFIG =================
void loadConfig() {
if (!sdReady) return;
spiToSD();
File f = sd.open("/ARGUS/CONFIG/config.txt");
if (f) {
while (f.available()) {
String line = f.readStringUntil('\n');
int eq = line.indexOf('=');
if (eq < 0) continue;
String k = line.substring(0, eq); k.trim();
String v = line.substring(eq + 1); v.trim();
if (k == "brightness") brightnessLevel = constrain(v.toInt(), 0, BRIGHTNESS_LEVELS - 1);
if (k == "buzzer") buzzerEnabled = (v == "1");
}
f.close();
}
spiIdle();
}
void saveConfig() {
if (!sdReady) return;
spiToSD();
File f = sd.open("/ARGUS/CONFIG/config.txt", FILE_WRITE);
if (f) {
f.printf("brightness=%d\n", brightnessLevel);
f.printf("buzzer=%d\n", buzzerEnabled ? 1 : 0);
f.close();
}
spiIdle();
}

// ================= GIF / BMP =================
void GIFDraw(GIFDRAW* pDraw) {
uint8_t* px = pDraw->pPixels;
uint16_t* pal = pDraw->pPalette;
int x = pDraw->iX, y = pDraw->iY + pDraw->y, w = pDraw->iWidth;
if (y < 0 || y >= 128) return;
spiToTFT();
for (int i = 0; i < w; i++) {
if (x + i < 0 || x + i >= 160) continue;
tft.drawPixel(x + i, y, pal[px[i]]);
}
spiIdle();
}
void* GIFOpenFile(const char* fname, int32_t* pSize) {
spiToSD();
File* f = new File(sd.open(fname, FILE_READ));
if (f && f) { pSize = f->size(); return (void)f; }
if (f) delete f;
spiIdle();
return nullptr;
}
void GIFCloseFile(void pHandle) {
File* f = (File*)pHandle;
if (f) { f->close(); delete f; }
spiIdle();
}
int32_t GIFReadFile(GIFFILE* pFile, uint8_t* pBuf, int32_t iLen) {
File* f = (File*)pFile->fHandle;
return f ? f->read(pBuf, iLen) : 0;
}
int32_t GIFSeekFile(GIFFILE* pFile, int32_t iPosition) {
File* f = (File*)pFile->fHandle;
if (!f) return 0;
f->seek(iPosition);
return iPosition;
}

uint16_t bpp = h[28] | (h[29]<<8);

struct ProbeEntry { uint8_t mac[6]; char ssid[24]; int8_t rssi; unsigned long lastSeen; };

volatile int8_t probeLastRSSI = 0;
int findProbe(const uint8_t* mac) {
for (int i = 0; i < probeCount; i++)
if (memcmp(probeList[i].mac, mac, 6) == 0) return i;
return -1;
}
void shiftProbes() {
for (int i = 0; i < PROBE_MAX - 1; i++) probeList[i] = probeList[i + 1];
probeCount = PROBE_MAX - 1;
}
void IRAM_ATTR probeCallback(void* buf, wifi_promiscuous_pkt_type_t type) {
if (type != WIFI_PKT_MGMT) return;
wifi_promiscuous_pkt_t* pkt = (wifi_promiscuous_pkt_t*)buf;
uint8_t* p = pkt->payload; int len = pkt->rx_ctrl.sig_len;
if (len < 24) return;
if ((p[0] & 0xF0) != 0x40) return;
probePackets++; probeLastSeen = millis(); probeLastRSSI = pkt->rx_ctrl.rssi;
memcpy((void*)probeLastMAC, p + 10, 6);
probeLastSSID[0] = '\0';
for (int i = 24; i < len - 2; ) {
uint8_t tid = p[i], tl = p[i + 1];
if (i + 2 + tl > len) break;
if (tid == 0x00) {
int l = tl > 23 ? 23 : tl;
memcpy((void*)probeLastSSID, p + i + 2, l);
probeLastSSID[l] = '\0'; break;
}
i += 2 + tl;
}
if (probeLastSSID[0] == '\0') {
strncpy((char*)probeLastSSID, "<hidden>", 23); probeLastSSID[23] = '\0';
}
}
void updateProbeList() {
if (probeLastSeen == 0) return;
int idx = findProbe((const uint8_t*)probeLastMAC);
if (idx < 0) {
if (probeCount >= PROBE_MAX) shiftProbes();
idx = probeCount++;
memcpy(probeList[idx].mac, (const void*)probeLastMAC, 6);
playSnifferPing();
}
strncpy(probeList[idx].ssid, (const char*)probeLastSSID, 23); probeList[idx].ssid[23] = '\0';
probeList[idx].rssi = probeLastRSSI; probeList[idx].lastSeen = probeLastSeen;
probeLastSeen = 0;
}

const int RADAR_CX = 80, RADAR_CY = 80, RADAR_MAX_R = 48;
struct RadarBlip { char ssid[20]; int8_t rssi; float angle, radius; bool open; uint8_t channel; bool valid; };
const int RADAR_MAX = 20;
RadarBlip radarBlips[RADAR_MAX];
int radarCount = 0, radarMode = 0;
unsigned long lastRadarScan = 0;
const int RADAR_SCAN_INTERVAL = 6000;
float rssiToRadius(int8_t rssi) {
const int RSSI_MIN = -95, RSSI_MAX = -30;
if (rssi < RSSI_MIN) rssi = RSSI_MIN; if (rssi > RSSI_MAX) rssi = RSSI_MAX;
float t = (float)(rssi - RSSI_MIN) / (float)(RSSI_MAX - RSSI_MIN);
return RADAR_MAX_R * (1.0 - t) + 6;
}
uint16_t rssiColor(int8_t rssi) { if (rssi > -50) return COL_ACCENT; if (rssi > -70) return COL_ACCENT2; return COL_DIM; }
void scanRadar() {
WiFi.scanDelete(); WiFi.scanNetworks(true, true);
unsigned long start = millis();
while (WiFi.scanComplete() == WIFI_SCAN_RUNNING && millis() - start < 8000) delay(150);
int n = WiFi.scanComplete(); if (n < 0) n = 0;
radarCount = 0;
for (int i = 0; i < n && radarCount < RADAR_MAX; i++) {
String ssid = WiFi.SSID(i); if (ssid.length() == 0) ssid = "<hidden>";
bool open = WiFi.encryptionType(i) == WIFI_AUTH_OPEN;
if (radarMode == 1 && !open) continue;
int8_t rssi = WiFi.RSSI(i);
strncpy(radarBlips[radarCount].ssid, ssid.c_str(), 19);
radarBlips[radarCount].ssid[19] = '\0';
radarBlips[radarCount].rssi = rssi;
radarBlips[radarCount].open = open;
radarBlips[radarCount].channel = WiFi.channel(i);
radarBlips[radarCount].radius = rssiToRadius(rssi);
uint8_t* bssid = WiFi.BSSID(i);
float seed = bssid ? (bssid[5] / 255.0f) : (i / 20.0f);
radarBlips[radarCount].angle = seed * 2 * PI;
radarBlips[radarCount].valid = true;
radarCount++;
}
}

void sendSavedIrFile(const char* name) {
char path[80]; snprintf(path, sizeof(path), "/ARGUS/CONFIG/IR/%s", name);
int proto = -1; uint64_t val = 0; int bits = 0;
spiToSD();
File f = sd.open(path);
if (f) {
while (f.available()) {
String line = f.readStringUntil('\n');
int eq = line.indexOf('='); if (eq < 0) continue;
String k = line.substring(0, eq); k.trim();
String v = line.substring(eq + 1); v.trim();
if (k == "PROTO") proto = v.toInt();
if (k == "VALUE") val = strtoull(v.c_str(), NULL, 0);
if (k == "BITS") bits = v.toInt();
}
f.close();
}
spiIdle();
if (proto < 0 || bits <= 0) { beepError(); return; }
uint32_t now = millis();
if ((uint32_t)(now - lastIrSend) < IR_SEND_GUARD_MS) return;
switch (proto) {
case NEC:      irsend.sendNEC(val, bits); break;
case SAMSUNG:  irsend.sendSAMSUNG(val, bits); break;
case LG:       irsend.sendLG(val, bits); break;
case SONY:     irsend.sendSony(val, bits, 2); break;
case PANASONIC:irsend.sendPanasonic64(val, bits); break;
case SHARP:    irsend.sendSharpRaw(val, bits); break;
default: beepError(); return;
}
lastIrSend = millis();
playIRCue();
}

// ================= WEATHER =================
float jsonFloat(const String& src, const char* key) {
int i = src.indexOf(key); if (i < 0) return 0;
i += strlen(key);
while (i < (int)src.length() && (src[i] == ' ' || src[i] == ':')) i++;
int e = i;
while (e < (int)src.length() && (isDigit(src[e]) || src[e] == '-' || src[e] == '.')) e++;
return src.substring(i, e).toFloat();
}
String jsonStr(const String& src, const char* key) {
int i = src.indexOf(key); if (i < 0) return "";
i = src.indexOf('"', i + strlen(key)); if (i < 0) return ""; i++;
int e = src.indexOf('"', i); if (e < 0) return "";
return src.substring(i, e);
}
bool fetchGeoAndTime() {
if (WiFi.status() != WL_CONNECTED) return false;
HTTPClient http; http.begin("http://ip-api.com/json/?fields=lat,lon,city,offset");
int code = http.GET(); if (code != 200) { http.end(); return false; }
String payload = http.getString(); http.end();
lastLat = jsonFloat(payload, ""lat":");
lastLon = jsonFloat(payload, ""lon":");
lastCity = jsonStr(payload, ""city":");
long offset = (long)jsonFloat(payload, ""offset":");
configTime(offset, 0, "pool.ntp.org", "time.nist.gov");
return true;
}
bool fetchWeather() {
if (WiFi.status() != WL_CONNECTED) return false;
NetworkClientSecure client; client.setInsecure();
HTTPClient https;
String url = "https://api.open-meteo.com/v1/forecast?latitude=" + String(lastLat, 4) +
"&longitude=" + String(lastLon, 4) + "&current_weather=true";
https.begin(client, url);
int code = https.GET(); if (code != 200) { https.end(); return false; }
String payload = https.getString(); https.end();
int cwIdx = payload.indexOf(""current_weather":");
String cw = cwIdx >= 0 ? payload.substring(cwIdx) : payload;
weatherData.tempC = jsonFloat(cw, ""temperature":");
weatherData.code = (int)jsonFloat(cw, ""weathercode":");
weatherData.valid = true;
return true;
}
const char* weatherCodeText(int code) {
if (code == 0) return "Clear";
if (code <= 3) return "Partly cloudy";
if (code == 45 || code == 48) return "Fog";
if (code <= 57) return "Drizzle";
if (code <= 67) return "Rain";
if (code <= 77) return "Snow";
if (code <= 82) return "Rain showers";
if (code <= 86) return "Snow showers";
if (code >= 95) return "Thunderstorm";
return "Unknown";
}

if (getLocalTime(&ti, 100)) {
strftime(ts, sizeof(ts), "%H:%M", &ti);
strftime(ds, sizeof(ds), "%d.%m.%Y", &ti);
}
centerText(ts, 30, COL_TEXT, 2);

}

Serial.begin(115200);
delay(200);

Serial.println("====================");

pinMode(BTN_BACK, INPUT_PULLUP);
pinMode(BTN_NEXT, INPUT_PULLUP);
pinMode(BTN_SELECT, INPUT_PULLUP);
pinMode(TFT_BL, OUTPUT);
pinMode(TFT_CS, OUTPUT);
pinMode(SD_CS, OUTPUT);
pinMode(BUZZER_PIN, OUTPUT);
pinMode(IR_PIN, OUTPUT);
digitalWrite(TFT_CS, HIGH);
digitalWrite(SD_CS, HIGH);
digitalWrite(IR_PIN, LOW);

// SPI: SCK=18, MISO=19, MOSI=23
SPI.begin(18, 19, 23, -1);

// Buzzer
ledcAttach(BUZZER_PIN, 2000, 10);
applyBrightness();

// TFT
tft.initR(INITR_BLACKTAB);
tft.setRotation(3);
tft.setTextWrap(false);
clearScreen();
header("ARGUS");
centerText("SYSTEM READY", 58, COL_ACCENT, 2);
centerText("v2.0", 82, COL_DIM);
Serial.println("TFT : OK");

// IR
irsend.begin();
irrecv.enableIRIn();
irrecv.disableIRIn(); // enable only when needed
Serial.println("IR  : OK");

// SD
sdReady = initSD();
if (sdReady) {
Serial.println("SD  : OK");
sdEnsureDirs();
loadConfig();
playSDBeep();
} else {
Serial.println("SD  : ERROR");
}

// WiFi
WiFi.mode(WIFI_STA);
WiFi.disconnect();
Serial.println("WIFI: READY");

// Buttons
Serial.println("BTN : OK");

Serial.println("====================");

randomSeed(micros());
playBootMelody();
delay(200);

currentScreenTransition(MAIN);
}

screen = s;
prevScreen = s;
screenEntered = true;
}

// ================= LOOP =================
void loop() {
bool entered = screenEntered;
screenEntered = false;

switch (screen) {
case MAIN:              tickMain(entered);              break;
case WIFI_MENU:         tickWifiMenu(entered);          break;
case WIFI_SCAN:         tickWifiScan(entered);          break;
case WIFI_INFO:         tickWifiInfo(entered);          break;
case WIFI_STATUS:       tickWifiStatus(entered);        break;
case WIFI_HOTSPOT:      tickHotspot(entered);           break;
case BEACON_SPAM:       tickBeaconSpam(entered);        break;
case CAPTIVE_PORTAL:    tickCaptivePortal(entered);     break;
case SNIFFER:           tickSniffer(entered);           break;
case DEAUTH:            tickDeauth(entered);            break;
case RADAR:             tickRadar(entered);             break;
case DEAUTH_DETECTOR:   tickDeauthDetector(entered);    break;
case KARMA:             tickKarma(entered);             break;

case IR_MENU:           tickIrMenu(entered);            break;  
case IR_TV_BRANDS:      tickIrBrands(entered);          break;  
case IR_TV_MODELS:      tickIrModels(entered);          break;  
case IR_TV_REMOTE:      tickIrRemote(entered);          break;  
case IR_RECEIVE:        tickIrReceive(entered);         break;  
case IR_SAVED:          tickIrSaved(entered);           break;  

case SD_MENU:           tickSDMenu(entered);            break;  
case SD_GIF:            tickSDFiles(entered, false);    break; // handled inside SD_FILES  
case SD_IMAGE:          tickSDFiles(entered, false);    break;  
case SD_FILES:          tickSDFiles(entered, false);    break;  
case SD_DELETE:         tickSDFiles(entered, true);     break;  

case DISPLAY_TEST:      tickDisplayTest(entered);       break;  
case WEATHER:           tickWeather(entered);           break;  
case INFO:              tickInfo(entered);              break;  
case SETTINGS:          tickSettings(entered);          break;  

default:                screen = MAIN;                  break;

}

if (screen != prevScreen) {
prevScreen = screen;
screenEntered = true;
}
delay(2);
}
